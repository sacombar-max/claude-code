'==============================================================================
' CUBICADOR AUTOMATICO
'
' Flujo diario:
'   1. CubicarExportSAP    -> lee el export de SAP, toma las entregas que NO
'                             estan en CRUCE, calcula cubicaje, valor y zona.
'                             Pregunta dos filtros: desde que fecha tomar las
'                             pendientes para vehiculo, y que remisiones van a
'                             bodega (por fecha o despues de la ultima registrada).
'                             Crea CUBICAR HOY, RESUMEN TRANSPORTE, ALERTAS y la
'                             vista previa de los correos (ArmarVistaPrevia).
'   2. Revisar/editar      -> PREVIA BODEGA y PREVIA VEHICULOS muestran todo lo que
'                             ira en los correos. En vehiculos se edita: Pedir
'                             (SI/NO), Tipo, Doc. Transporte, Nota; se pueden
'                             borrar lineas. Los vehiculos salen por zona, divididos
'                             si no caben (m3 y maximo de descargues: TURBO 13,
'                             SENCILLO 10, PATINETA/TRACTOMULA 1); si hay carga
'                             para Cucuta, Cucuta + Bucaramanga van solas.
'   3. EnviarCorteBodega   -> borradores a bodega desde PREVIA BODEGA, con TXT.
'   4. SolicitarVehiculo   -> borradores de solicitud desde PREVIA VEHICULOS.
'   ArmarVistaPrevia       -> rehace las vistas (ej. despues de filtrar CUBICAR HOY).
'   MarcarRemisionesCubicadas -> marca en CRUCE las filas visibles (manual).
'
' Ningun correo se envia solo: todos quedan como borrador para revisar.
'
' Hojas que usa del cubicador:
'   MATERIAL         A = IP, C = volumen teorico m3/und
'   VALO MATERIAL    C = IP, D = valor sin IVA (desde la fila 4)
'   llanta_zona      J = ciudad, K = zona
'   CRUCE            A = remisiones ya cubicadas (se agregan en el paso 3)
'   CUBICAJE BODEGA  A3:C6 = etiqueta (ej. "42 m3"), tipo de vehiculo, m3 maximo
'
' Hojas que crea: CUBICAR HOY, RESUMEN TRANSPORTE, ALERTAS, HISTORIAL BODEGA
'==============================================================================
Option Explicit

' --- Configuracion: destinatarios ------------------------------------------
' Separe con punto y coma. Puede usar correos o nombres como aparecen en Outlook.
Private Const PARA_TRANSPORTE As String = ""
Private Const CC_TRANSPORTE As String = ""
Private Const PARA_BODEGA As String = ""
Private Const CC_BODEGA As String = ""

' --- Configuracion: otros ----------------------------------------------------
Private Const CARPETA_TXT As String = ""            ' carpeta donde SAP deja los TXT de remision; vacio = se pregunta
Private Const TOPE_SATURACION As Double = 1#        ' solo para el vehiculo sugerido del resumen
Private Const CENTRO_CUBICADOR As String = "CDLI"

Private Const HOJA_DETALLE As String = "CUBICAR HOY"
Private Const HOJA_RESUMEN As String = "RESUMEN TRANSPORTE"
Private Const HOJA_ALERTAS As String = "ALERTAS"
Private Const HOJA_HIST As String = "HISTORIAL BODEGA"
Private Const HOJA_PREV_BOD As String = "PREVIA BODEGA"
Private Const HOJA_PREV_VEH As String = "PREVIA VEHICULOS"
Private Const SIN_ZONA As String = "SIN ZONA"

' Columnas de CUBICAR HOY
Private Const C_FECHA As Long = 1, C_CORTE As Long = 2, C_CENTRO As Long = 3, C_REM As Long = 4
Private Const C_CLIENTE As Long = 5, C_CIUDAD As Long = 6, C_ZONA As Long = 7, C_COD As Long = 8
Private Const C_DESC As Long = 9, C_CANT As Long = 10, C_PESO As Long = 11, C_CODSAP As Long = 12
Private Const C_DIR As Long = 13, C_CUBUND As Long = 14, C_CUB As Long = 15, C_EMP As Long = 16
Private Const C_VALOR As Long = 17, C_NUEVO As Long = 18

' Vehiculos (se leen de CUBICAJE BODEGA!A3:C6)
Private vehNombre() As String
Private vehCap() As Double
Private vehEtiqueta() As String

'==============================================================================
' 1. CUBICAR EXPORT DE SAP
'==============================================================================
Public Sub CubicarExportSAP()
    Dim archivo As Variant, wbSAP As Workbook, datos As Variant
    Dim dMat As Object, dVal As Object, dZona As Object, dCruce As Object
    Dim alertas As Collection

    archivo = Application.GetOpenFilename("Excel (*.xls*),*.xls*", , "Seleccione el export de SAP del dia")
    If VarType(archivo) = vbBoolean Then Exit Sub

    Dim corte As String
    corte = InputBox("Texto de la Fecha Corte:", "Fecha de corte", "1 Corte " & MesEnEspanol(Date) & " " & Format(Date, "dd"))
    If Len(Trim(corte)) = 0 Then Exit Sub

    Dim dHist As Object: Set dHist = Diccionario(HojaHistorial(), 1, 2, 2)

    ' Filtro 1: desde que fecha se consideran las remisiones pendientes para vehiculo
    Dim txt As String, desdeVeh As Date, hayDesdeVeh As Boolean
    txt = InputBox("PENDIENTES PARA VEHICULO" & vbCrLf & _
                   "Tomar remisiones con fecha de remision desde (dd/mm/aaaa)." & vbCrLf & _
                   "Deje vacio para tomar todas las pendientes (no estan en CRUCE):", "Filtro vehiculos", "")
    If Len(Trim(txt)) > 0 Then
        If Not IsDate(txt) Then MsgBox "Fecha no valida: " & txt, vbExclamation: Exit Sub
        desdeVeh = CDate(txt): hayDesdeVeh = True
    End If

    ' Filtro 2: que remisiones nuevas van al correo de bodega
    Dim modoBod As String, bodDesde As Date, bodHasta As Date, bodUltRem As Double, kh As Variant, txtFiltro As String
    For Each kh In dHist.Keys
        If IsNumeric(kh) Then
            If CDbl(kh) > bodUltRem Then bodUltRem = CDbl(kh)
        End If
    Next kh
    modoBod = InputBox("CORREO A BODEGA - que remisiones nuevas incluir:" & vbCrLf & vbCrLf & _
                       "1 = por fecha de remision (desde / hasta)" & vbCrLf & _
                       "2 = posteriores a la ultima remision registrada (" & Format(bodUltRem, "0") & ")" & vbCrLf & _
                       "3 = sin filtro", "Filtro bodega", "1")
    Select Case Trim(modoBod)
        Case "1"
            txt = InputBox("Fecha de remision DESDE (dd/mm/aaaa):", "Filtro bodega", Format(Date, "dd/mm/yyyy"))
            If Not IsDate(txt) Then Exit Sub
            bodDesde = CDate(txt)
            txt = InputBox("Fecha de remision HASTA (dd/mm/aaaa):", "Filtro bodega", Format(Date, "dd/mm/yyyy"))
            If Not IsDate(txt) Then Exit Sub
            bodHasta = CDate(txt)
            txtFiltro = "fecha de remision del " & Format(bodDesde, "dd/mm/yyyy") & " al " & Format(bodHasta, "dd/mm/yyyy")
        Case "2"
            txt = InputBox("Incluir remisiones con numero MAYOR a:", "Filtro bodega", Format(bodUltRem, "0"))
            If Not IsNumeric(txt) Then Exit Sub
            bodUltRem = CDbl(txt)
            txtFiltro = "remisiones mayores a " & Format(bodUltRem, "0")
        Case "3"
            txtFiltro = "sin filtro"
        Case Else
            Exit Sub
    End Select

    Application.ScreenUpdating = False
    Application.Calculation = xlCalculationManual
    On Error GoTo Fallo

    LeerVehiculos
    Set dMat = Diccionario(ThisWorkbook.Worksheets("MATERIAL"), 1, 3, 2)
    Set dVal = Diccionario(ThisWorkbook.Worksheets("VALO MATERIAL"), 3, 4, 4)
    Set dZona = DiccionarioCiudades(ThisWorkbook.Worksheets("llanta_zona"), 10, 11, 2)
    Set dCruce = Diccionario(ThisWorkbook.Worksheets("CRUCE"), 1, 1, 2)
    Set alertas = New Collection

    Set wbSAP = Workbooks.Open(archivo, ReadOnly:=True)
    datos = wbSAP.Worksheets(1).UsedRange.Value
    wbSAP.Close SaveChanges:=False

    ' Columnas del export (se busca por nombre; si un nombre se repite se toma la primera)
    Dim cFecha&, cEntrega&, cCliente&, cCiudad&, cMaterial&, cDesc&, cCant&, cPeso&, cCodSAP&, cDir&
    cFecha = Col(datos, "Fec.Entrega"):        cEntrega = Col(datos, "Entrega")
    cCliente = Col(datos, "Nombre Recibid"):   cCiudad = Col(datos, "Ciudad Recibidor")
    cMaterial = Col(datos, "Material"):        cDesc = Col(datos, "Descripc.Material")
    cCant = Col(datos, "Cantidad Entrega"):    cPeso = Col(datos, "Peso Bruto")
    cCodSAP = Col(datos, "Cold.Recibidor"):    cDir = Col(datos, "Dir.Entrega")

    ' --- Detalle -------------------------------------------------------------
    Const NC As Long = 18
    Dim out() As Variant, n As Long, i As Long
    ReDim out(1 To UBound(datos, 1), 1 To NC)
    Dim yaCruce As Long, nuevas As Object: Set nuevas = CreateObject("Scripting.Dictionary")
    Dim yaContada As Object: Set yaContada = CreateObject("Scripting.Dictionary")
    Dim fueraVeh As Object: Set fueraVeh = CreateObject("Scripting.Dictionary")
    Dim paraBodega As Object: Set paraBodega = CreateObject("Scripting.Dictionary")
    Dim pasa As Boolean

    For i = 2 To UBound(datos, 1)
        Dim rem_ As String, cant As Double, mat As String, ciudad As String, zona As String
        rem_ = Clave(datos(i, cEntrega))
        cant = Num(datos(i, cCant))
        If Len(rem_) > 0 And cant > 0 Then
            If dCruce.Exists(rem_) Then
                If Not yaContada.Exists(rem_) Then yaContada.Add rem_, True: yaCruce = yaCruce + 1
            ElseIf hayDesdeVeh And FechaAntes(datos(i, cFecha), desdeVeh) Then
                fueraVeh(rem_) = True
            Else
                n = n + 1
                nuevas(rem_) = True
                mat = Clave(datos(i, cMaterial))
                ciudad = Trim(CStr(datos(i, cCiudad)))

                If dZona.Exists(NormCiudad(ciudad)) Then
                    zona = dZona(NormCiudad(ciudad))
                Else
                    zona = SIN_ZONA
                    Alerta alertas, "Ciudad sin zona en llanta_zona", ciudad, rem_
                End If
                If Not dMat.Exists(mat) Then Alerta alertas, "Material sin volumen en MATERIAL (cubicaje = 0)", mat & " " & datos(i, cDesc), rem_
                If Not dVal.Exists(mat) Then Alerta alertas, "Material sin valor en VALO MATERIAL", mat & " " & datos(i, cDesc), rem_

                out(n, 1) = datos(i, cFecha)
                out(n, 2) = IIf(dHist.Exists(rem_), dHist(rem_), corte)
                out(n, 3) = CENTRO_CUBICADOR
                out(n, 4) = CDbl(rem_)
                out(n, 5) = datos(i, cCliente)
                out(n, 6) = ciudad
                out(n, 7) = zona
                out(n, 8) = Val(mat)
                out(n, 9) = datos(i, cDesc)
                out(n, 10) = cant
                out(n, 11) = Num(datos(i, cPeso))
                out(n, 12) = datos(i, cCodSAP)
                out(n, 13) = datos(i, cDir)
                out(n, 14) = IIf(dMat.Exists(mat), Num(dMat(mat)), 0)
                out(n, 15) = out(n, 14) * cant
                out(n, 16) = IIf(Left(rem_, 2) = "29", "PIRELLI TYRE", "PROMETEON")
                out(n, 17) = IIf(dVal.Exists(mat), Num(dVal(mat)), 0) * cant
                If dHist.Exists(rem_) Then
                    out(n, 18) = "NO"
                Else
                    Select Case Trim(modoBod)
                        Case "1": pasa = Not FechaAntes(datos(i, cFecha), bodDesde) And Not FechaDespues(datos(i, cFecha), bodHasta) And IsDate(datos(i, cFecha))
                        Case "2": pasa = (CDbl(rem_) > bodUltRem)
                        Case Else: pasa = True
                    End Select
                    out(n, 18) = IIf(pasa, "SI", "NO - fuera de filtro")
                    If pasa Then paraBodega(rem_) = True
                End If
            End If
        End If
    Next i

    If n = 0 Then
        Application.Calculation = xlCalculationAutomatic
        Application.ScreenUpdating = True
        MsgBox "No hay remisiones para cubicar." & vbCrLf & "Ya en CRUCE: " & yaCruce & vbCrLf & _
               "Fuera del filtro de fecha para vehiculo: " & fueraVeh.Count, vbInformation
        Exit Sub
    End If

    Dim wsD As Worksheet: Set wsD = HojaNueva(HOJA_DETALLE)
    wsD.Range("A1").Resize(1, NC).Value = Array("Fecha Remision", "Fecha Corte", "CENTRO", "Remision", "Cliente", _
        "Ciudad", "Zona", "Codigo", "Descripcion", "Cantidad", "Peso (kg)", "Codigo SAP", "Direccion", _
        "Cubicaje/und", "Cubicaje Total", "EMPRESA", "Valor total sin IVA", "Nuevo en corte")
    wsD.Range("A2").Resize(n, NC).Value = out
    wsD.Columns("A").NumberFormat = "dd/mm/yyyy"
    wsD.Columns("D").NumberFormat = "0"
    wsD.Columns("N:O").NumberFormat = "0.000"
    wsD.Columns("Q").NumberFormat = "#,##0"
    FormatoTabla wsD, n + 1, NC

    ' --- Resumen ---------------------------------------------------------------
    ArmarResumen out, n

    ' --- Alertas ---------------------------------------------------------------
    Dim wsA As Worksheet: Set wsA = HojaNueva(HOJA_ALERTAS)
    wsA.Range("A1:C1").Value = Array("Alerta", "Dato", "Remision")
    If alertas.Count = 0 Then
        wsA.Range("A2").Value = "Sin alertas"
    Else
        For i = 1 To alertas.Count
            wsA.Cells(i + 1, 1).Resize(1, 3).Value = alertas(i)
        Next i
    End If
    FormatoTabla wsA, alertas.Count + 1, 3

    Application.Calculation = xlCalculationAutomatic
    Application.ScreenUpdating = True
    ArmarVistaPrevia

    MsgBox "Remisiones pendientes para vehiculo: " & nuevas.Count & " (" & n & " lineas)" & vbCrLf & _
           "Ya cubicadas antes (en CRUCE): " & yaCruce & vbCrLf & _
           "Fuera del filtro de fecha para vehiculo: " & fueraVeh.Count & vbCrLf & _
           "Nuevas para el correo a bodega: " & paraBodega.Count & " (" & txtFiltro & ")" & vbCrLf & _
           "Alertas: " & alertas.Count & IIf(alertas.Count > 0, "  -> revise la hoja " & HOJA_ALERTAS, "") & vbCrLf & vbCrLf & _
           "Revise " & HOJA_PREV_BOD & " y " & HOJA_PREV_VEH & " antes de armar los correos.", _
           IIf(alertas.Count > 0, vbExclamation, vbInformation), "Cubicaje listo"
    Exit Sub

Fallo:
    Dim msgError As String: msgError = Err.Description
    Application.Calculation = xlCalculationAutomatic
    Application.ScreenUpdating = True
    On Error Resume Next
    If Not wbSAP Is Nothing Then wbSAP.Close SaveChanges:=False
    MsgBox "Error: " & msgError, vbCritical, "Cubicador"
End Sub

' Resumen por zona y por ciudad: m3, kilos, unidades, valor, saturacion y vehiculo
Private Sub ArmarResumen(out As Variant, n As Long)
    Dim grpZona As Object, grpCiudad As Object, i As Long
    Set grpZona = CreateObject("Scripting.Dictionary")
    Set grpCiudad = CreateObject("Scripting.Dictionary")

    For i = 1 To n
        Acumular grpZona, CStr(out(i, 7)), "", out, i
        Acumular grpCiudad, CStr(out(i, 7)) & "|" & NormCiudad(CStr(out(i, 6))), NormCiudad(CStr(out(i, 6))), out, i
    Next i

    Dim ws As Worksheet: Set ws = HojaNueva(HOJA_RESUMEN)
    Dim enc As Variant, k As Long, nv As Long
    nv = UBound(vehNombre)
    ReDim enc(1 To 7 + nv + 2)
    enc(1) = "Zona": enc(2) = "Ciudad": enc(3) = "Remisiones": enc(4) = "Unidades"
    enc(5) = "Kilos": enc(6) = "Cubicaje m3": enc(7) = "Valor sin IVA"
    For k = 1 To nv: enc(7 + k) = "% " & vehNombre(k) & " (" & vehCap(k) & " m3)": Next k
    enc(8 + nv) = "Vehiculo sugerido"
    enc(9 + nv) = "Descargues"

    Dim fila As Long: fila = 1
    ws.Cells(fila, 1).Value = "POR ZONA": ws.Cells(fila, 1).Font.Bold = True
    fila = fila + 1
    fila = EscribirGrupos(ws, fila, enc, grpZona, False)

    fila = fila + 2
    ws.Cells(fila, 1).Value = "POR CIUDAD": ws.Cells(fila, 1).Font.Bold = True
    fila = fila + 1
    fila = EscribirGrupos(ws, fila, enc, grpCiudad, True)

    ws.Columns.AutoFit
End Sub

Private Sub Acumular(d As Object, clave As String, ciudad As String, out As Variant, i As Long)
    Dim g As Variant
    If Not d.Exists(clave) Then
        ' zona, ciudad, remisiones(dict), unidades, kilos, m3, valor
        d.Add clave, Array(CStr(out(i, 7)), ciudad, CreateObject("Scripting.Dictionary"), 0#, 0#, 0#, 0#, CreateObject("Scripting.Dictionary"))
    End If
    g = d(clave)
    g(2).Item(CStr(out(i, 4))) = True
    g(3) = g(3) + out(i, 10)
    g(4) = g(4) + out(i, 11)
    g(5) = g(5) + out(i, 15)
    g(6) = g(6) + out(i, 17)
    g(7).Item(NormCiudad(CStr(out(i, 6))) & "|" & NormCiudad(CStr(out(i, 13)))) = True
    d(clave) = g
End Sub

Private Function EscribirGrupos(ws As Worksheet, fila As Long, enc As Variant, d As Object, conCiudad As Boolean) As Long
    Dim nv As Long, k As Long, c As Long, key As Variant, g As Variant, ini As Long
    Dim claves As Variant
    nv = UBound(vehNombre)
    ini = fila
    ws.Cells(fila, 1).Resize(1, UBound(enc)).Value = enc
    claves = Ordenadas(d)
    For Each key In claves
        g = d(key)
        fila = fila + 1
        ws.Cells(fila, 1).Value = g(0)
        ws.Cells(fila, 2).Value = IIf(conCiudad, g(1), "(todas)")
        ws.Cells(fila, 3).Value = g(2).Count
        ws.Cells(fila, 4).Value = g(3)
        ws.Cells(fila, 5).Value = g(4)
        ws.Cells(fila, 6).Value = g(5)
        ws.Cells(fila, 7).Value = g(6)
        For k = 1 To nv
            ws.Cells(fila, 7 + k).Value = g(5) / vehCap(k)
        Next k
        ws.Cells(fila, 8 + nv).Value = VehiculoSugerido(CDbl(g(5)), CLng(g(7).Count))
        ws.Cells(fila, 9 + nv).Value = g(7).Count
    Next key
    ws.Range(ws.Cells(ini + 1, 5), ws.Cells(fila, 5)).NumberFormat = "#,##0.0"
    ws.Range(ws.Cells(ini + 1, 6), ws.Cells(fila, 6)).NumberFormat = "0.00"
    ws.Range(ws.Cells(ini + 1, 7), ws.Cells(fila, 7)).NumberFormat = "#,##0"
    With ws.Range(ws.Cells(ini + 1, 8), ws.Cells(fila, 7 + nv))
        .NumberFormat = "0%"
        .FormatConditions.Delete
        .FormatConditions.Add(xlCellValue, xlGreater, TOPE_SATURACION).Interior.Color = RGB(255, 199, 206)
    End With
    With ws.Range(ws.Cells(ini, 1), ws.Cells(ini, UBound(enc)))
        .Font.Bold = True
        .Interior.Color = RGB(255, 255, 0)
    End With
    ws.Range(ws.Cells(ini, 1), ws.Cells(fila, UBound(enc))).Borders.LineStyle = xlContinuous
    EscribirGrupos = fila
End Function

' Regla: el vehiculo mas pequeno cuya saturacion sea <= TOPE_SATURACION.
' Si no cabe en el mas grande, se indica cuantos del mas grande se necesitan.
Private Function VehiculoSugerido(m3 As Double, pts As Long) As String
    Dim k As Long
    If m3 <= 0 Then VehiculoSugerido = "Sin cubicaje (revisar ALERTAS)": Exit Function
    If Factible(m3, pts) Then
        k = IndiceSugerido(m3, pts)
        VehiculoSugerido = vehNombre(k) & " (" & Format(m3 / vehCap(k), "0%") & ")"
    Else
        VehiculoSugerido = "Dividir en varios vehiculos (" & pts & " descargues)"
    End If
End Function

'==============================================================================
' 2. VISTA PREVIA: arma en Excel lo que ira en los correos (se revisa y edita)
'    - PREVIA BODEGA:    remisiones nuevas del corte, agrupadas por cliente, con TXT
'    - PREVIA VEHICULOS: un bloque por vehiculo (por zona, dividido si no cabe)
'    Se ejecuta sola al final de CubicarExportSAP. Si filtra CUBICAR HOY,
'    vuelva a ejecutar ArmarVistaPrevia para rehacer el plan de vehiculos.
'==============================================================================
Public Sub ArmarVistaPrevia()
    Dim ws As Worksheet
    Set ws = HojaDetalle(): If ws Is Nothing Then Exit Sub
    Application.ScreenUpdating = False
    LeerVehiculos
    PreviaBodega ws
    PreviaVehiculos ws
    Application.ScreenUpdating = True
    ThisWorkbook.Worksheets(HOJA_PREV_VEH).Activate
End Sub

Private Sub PreviaBodega(ws As Worksheet)
    Dim pb As Worksheet, i As Long, ultima As Long, emp As String, corte As String
    Dim porEmpresa As Object: Set porEmpresa = CreateObject("Scripting.Dictionary")
    ultima = ws.Cells(ws.Rows.Count, C_REM).End(xlUp).Row
    For i = 2 To ultima
        If ws.Cells(i, C_NUEVO).Value = "SI" Then
            emp = CStr(ws.Cells(i, C_EMP).Value)
            If Not porEmpresa.Exists(emp) Then porEmpresa.Add emp, New Collection
            porEmpresa(emp).Add i
            corte = CStr(ws.Cells(i, C_CORTE).Value)
        End If
    Next i

    Set pb = HojaNueva(HOJA_PREV_BOD)
    pb.Range("A1").Value = "VISTA PREVIA - CORREO A BODEGA (revise y edite; luego ejecute EnviarCorteBodega)"
    pb.Range("A1").Font.Bold = True: pb.Range("A1").Font.Size = 13
    pb.Range("A2").Value = "Para": pb.Range("B2").Value = PARA_BODEGA
    pb.Range("A3").Value = "CC": pb.Range("B3").Value = CC_BODEGA
    pb.Range("A2:A3").Font.Bold = True
    If porEmpresa.Count = 0 Then
        pb.Range("A5").Value = "No hay remisiones nuevas para enviar a bodega."
        Exit Sub
    End If

    Dim txt As Object, carpeta As String
    carpeta = CarpetaTxt()
    If Len(carpeta) > 0 Then Set txt = ArchivosTxtPorRemision(carpeta) Else Set txt = CreateObject("Scripting.Dictionary")

    Dim fila As Long, k As Variant, ini As Long, r As Long, grupo As Collection
    fila = 5
    For Each k In porEmpresa.Keys
        pb.Cells(fila, 1).Value = "EMPRESA": pb.Cells(fila, 2).Value = k
        pb.Range(pb.Cells(fila, 1), pb.Cells(fila, 13)).Interior.Color = RGB(255, 230, 153)
        pb.Range(pb.Cells(fila, 1), pb.Cells(fila, 13)).Font.Bold = True
        pb.Cells(fila + 1, 1).Value = "Asunto": pb.Cells(fila + 1, 2).Value = AsuntoCorte(corte, CStr(k))
        pb.Cells(fila + 2, 1).Value = "Texto": pb.Cells(fila + 2, 2).Value = "Adjunto se encuentra el " & TextoCorte(corte) & ":"
        pb.Cells(fila + 3, 1).Value = "Corte": pb.Cells(fila + 3, 2).Value = corte
        pb.Range(pb.Cells(fila + 1, 1), pb.Cells(fila + 3, 1)).Font.Bold = True
        ini = fila + 4
        Set grupo = porEmpresa(k)
        fila = EscribirLineas(pb, ini, ws, grupo, False)
        pb.Cells(ini, 13).Value = "TXT"
        For r = ini + 1 To fila - 1
            If txt.Exists(Clave(pb.Cells(r, 3).Value)) Then
                pb.Cells(r, 13).Value = "OK"
            Else
                pb.Cells(r, 13).Value = "FALTA"
                pb.Cells(r, 13).Interior.Color = RGB(255, 199, 206)
            End If
        Next r
        pb.Range(pb.Cells(ini, 1), pb.Cells(fila - 1, 13)).Borders.LineStyle = xlContinuous
        fila = fila + 1
    Next k
    pb.Columns("A:M").AutoFit
    pb.Columns("B").ColumnWidth = Application.Max(pb.Columns("B").ColumnWidth, 14)
End Sub

Private Sub PreviaVehiculos(ws As Worksheet)

' Escribe encabezado + lineas (agrupadas por Codigo SAP) a partir de la fila ini.
' Columnas A:M como en el correo (M = valor si conValor); N = m3 (solo en la vista).
' Devuelve la primera fila libre despues de las lineas.
Private Function EscribirLineas(dst As Worksheet, ini As Long, ws As Worksheet, filas As Collection, conValor As Boolean) As Long
    Dim titulos As Variant, orden As Collection, i As Long, r As Long, f As Long
    titulos = Array("Fecha Remisión", "CENTRO", "Remisión", "Cliente", "Ciudad", "Código", "Descripción", _
                    "Cantidad", "Peso", "Codigo SAP", "Dirección", "EMPRESA")
    dst.Cells(ini, 1).Resize(1, 12).Value = titulos
    If conValor Then dst.Cells(ini, 13).Value = "VALOR TOTAL SIN IVA": dst.Cells(ini, 14).Value = "m3"
    With dst.Range(dst.Cells(ini, 1), dst.Cells(ini, IIf(conValor, 14, 13)))
        .Font.Bold = True: .Interior.Color = RGB(255, 255, 0)
    End With
    Set orden = OrdenAgrupado(ws, filas)
    r = ini
    For i = 1 To orden.Count
        r = r + 1: f = orden(i)
        dst.Cells(r, 1).Value = ws.Cells(f, C_FECHA).Value
        dst.Cells(r, 2).Value = ws.Cells(f, C_CENTRO).Value
        dst.Cells(r, 3).Value = ws.Cells(f, C_REM).Value
        dst.Cells(r, 4).Value = ws.Cells(f, C_CLIENTE).Value
        dst.Cells(r, 5).Value = ws.Cells(f, C_CIUDAD).Value
        dst.Cells(r, 6).Value = ws.Cells(f, C_COD).Value
        dst.Cells(r, 7).Value = ws.Cells(f, C_DESC).Value
        dst.Cells(r, 8).Value = ws.Cells(f, C_CANT).Value
        dst.Cells(r, 9).Value = ws.Cells(f, C_PESO).Value
        dst.Cells(r, 10).Value = ws.Cells(f, C_CODSAP).Value
        dst.Cells(r, 11).Value = ws.Cells(f, C_DIR).Value
        dst.Cells(r, 12).Value = ws.Cells(f, C_EMP).Value
        If conValor Then
            dst.Cells(r, 13).Value = ws.Cells(f, C_VALOR).Value
            dst.Cells(r, 14).Value = ws.Cells(f, C_CUB).Value
        End If
    Next i
    If r > ini Then
        dst.Range(dst.Cells(ini + 1, 1), dst.Cells(r, 1)).NumberFormat = "d/mm/yyyy"
        dst.Range(dst.Cells(ini + 1, 3), dst.Cells(r, 3)).NumberFormat = "0"
        dst.Range(dst.Cells(ini + 1, 6), dst.Cells(r, 6)).NumberFormat = "0"
        dst.Range(dst.Cells(ini + 1, 8), dst.Cells(r, 8)).NumberFormat = "#,##0"
        dst.Range(dst.Cells(ini + 1, 9), dst.Cells(r, 9)).NumberFormat = "#,##0.000"
        dst.Range(dst.Cells(ini + 1, 10), dst.Cells(r, 10)).NumberFormat = "0"
        If conValor Then
            dst.Range(dst.Cells(ini + 1, 13), dst.Cells(r, 13)).NumberFormat = "$ #,##0.00"
            dst.Range(dst.Cells(ini + 1, 14), dst.Cells(r, 14)).NumberFormat = "0.000"
        End If
    End If
    EscribirLineas = r + 1
End Function

' Totales de un grupo de filas de CUBICAR HOY
Private Sub ResumenFilas(ws As Worksheet, filas As Collection, m3 As Double, pts As Long, destinos As String)
    Dim i As Long, ciudad As String, p As Object
    Set p = CreateObject("Scripting.Dictionary")
    m3 = 0: destinos = ""
    For i = 1 To filas.Count
        m3 = m3 + Num(ws.Cells(filas(i), C_CUB).Value)
        p(PuntoEntrega(ws, filas(i))) = True
        ciudad = UCase(Trim(CStr(ws.Cells(filas(i), C_CIUDAD).Value)))
        If InStr(1, " / " & destinos & " / ", " / " & ciudad & " / ") = 0 Then
            destinos = destinos & IIf(Len(destinos) > 0, " / ", "") & ciudad
        End If
    Next i
    pts = p.Count
End Sub

Public Sub EnviarCorteBodega()
    Dim pb As Worksheet, r As Long, ultima As Long, ini As Long, fin As Long
    On Error Resume Next
    Set pb = ThisWorkbook.Worksheets(HOJA_PREV_BOD)
    On Error GoTo 0

    Dim carpeta As String, txt As Object, adj As Collection, vistas As Object, rem_ As String
    Dim enviadas As Object: Set enviadas = CreateObject("Scripting.Dictionary")
    Dim sinTxt As String, n As Long
    carpeta = CarpetaTxt()
    If Len(carpeta) > 0 Then Set txt = ArchivosTxtPorRemision(carpeta) Else Set txt = CreateObject("Scripting.Dictionary")

    ultima = pb.Cells(pb.Rows.Count, 1).End(xlUp).Row
    For r = 1 To ultima
        If pb.Cells(r, 1).Value = "EMPRESA" Then

                Set adj = New Collection
                Set vistas = CreateObject("Scripting.Dictionary")
                Dim i As Long
                For i = ini + 1 To fin
                    rem_ = Clave(pb.Cells(i, 3).Value)
                    If Len(rem_) > 0 And Not vistas.Exists(rem_) Then
                        vistas.Add rem_, True
                        enviadas(rem_) = CStr(pb.Cells(r + 3, 2).Value)

                    TablaHtmlRango(pb, ini, fin, 12), adj

            End If

        End If

              "(Si responde No, volveran a salir en el proximo corte.)", vbYesNo + vbQuestion) = vbYes Then

    End If
End Sub

'==============================================================================
' 4. SOLICITUD DE VEHICULOS (desde la hoja PREVIA VEHICULOS)
'==============================================================================
Public Sub SolicitarVehiculo()
    Dim pv As Worksheet, r As Long, ultima As Long, ini As Long, fin As Long, i As Long
    On Error Resume Next
    Set pv = ThisWorkbook.Worksheets(HOJA_PREV_VEH)
    On Error GoTo 0
    If pv Is Nothing Then MsgBox "Primero ejecute CubicarExportSAP (o ArmarVistaPrevia).", vbExclamation: Exit Sub
    LeerVehiculos
    Application.Calculate

    Dim todas As Object: Set todas = CreateObject("Scripting.Dictionary")
    Dim hechas As String, omitidas As String, k As Long, tipo As String
    Dim unid As Double, kg As Double, m3 As Double, destinos As String, ciudad As String, rem_ As String
    Dim puntos As Object, rems As Object, numVeh As String, docTransp As String, extra As String, cuerpo As String

    ultima = pv.Cells(pv.Rows.Count, 1).End(xlUp).Row
    For r = 1 To ultima
        If pv.Cells(r, 1).Value = "VEHICULO No" Then
            ini = r + 2: fin = ini
            Do While Len(Trim(CStr(pv.Cells(fin + 1, 1).Value))) > 0 And pv.Cells(fin + 1, 1).Value <> "VEHICULO No"
                fin = fin + 1
            Loop
            numVeh = CStr(pv.Cells(r, 2).Value)
            If UCase(Trim(CStr(pv.Cells(r, 6).Value))) = "SI" And fin > ini Then
                ' Tipo de vehiculo elegido
                tipo = Trim(CStr(pv.Cells(r, 8).Value)): k = 0
                For i = 1 To UBound(vehEtiqueta)
                    If vehEtiqueta(i) = tipo Then k = i
                Next i
                ' Totales recalculados desde las lineas (por si se editaron)
                Set puntos = CreateObject("Scripting.Dictionary"): Set rems = CreateObject("Scripting.Dictionary")
                unid = 0: kg = 0: m3 = 0: destinos = ""
                For i = ini + 1 To fin
                    rem_ = Clave(pv.Cells(i, 3).Value)
                    If Len(rem_) > 0 Then rems(rem_) = True
                    unid = unid + Num(pv.Cells(i, 8).Value)
                    kg = kg + Num(pv.Cells(i, 9).Value)
                    m3 = m3 + Num(pv.Cells(i, 14).Value)
                    ciudad = UCase(Trim(CStr(pv.Cells(i, 5).Value)))
                    puntos(NormCiudad(ciudad) & "|" & NormCiudad(CStr(pv.Cells(i, 11).Value))) = True
                    If InStr(1, " / " & destinos & " / ", " / " & ciudad & " / ") = 0 Then
                        destinos = destinos & IIf(Len(destinos) > 0, " / ", "") & ciudad
                    End If
                Next i

                If k = 0 Then
                    omitidas = omitidas & vbCrLf & "Vehiculo " & numVeh & ": tipo '" & tipo & "' no valido"
                ElseIf puntos.Count > MaxDescargues(vehNombre(k)) Then
                    omitidas = omitidas & vbCrLf & "Vehiculo " & numVeh & ": " & puntos.Count & " descargues, " & _
                               vehNombre(k) & " permite " & MaxDescargues(vehNombre(k))
                Else
                    docTransp = Trim(CStr(pv.Cells(r, 10).Value))
                    extra = Trim(CStr(pv.Cells(r, 12).Value))
                    cuerpo = "<p>Buen d&iacute;a</p>" & _
                        IIf(Len(Trim(CStr(pv.Range("B4").Value))) > 0, "<p><b>NOTA: " & Html(CStr(pv.Range("B4").Value)) & "</b></p>", "") & _
                        "<table style='font-family:Calibri;font-size:11pt'>" & _
                        FilaDato("Tipo Veh&iacute;culo", Html(vehEtiqueta(k))) & _
                        FilaDato("Destino", Html(destinos)) & _
                        FilaDato("Doc. Transporte", Html(docTransp)) & _
                        FilaDato("Observaciones:", "Unidades cargadas " & Format(unid, "#,##0") & " - Puntos de entrega " & puntos.Count) & _
                        FilaDato("Nota:", "El veh&iacute;culo se encuentra " & Format(m3 / vehCap(k), "0%") & " de su capacidad " & Html(extra)) & _
                        FilaDato("", "Despachar seg&uacute;n TXT al menos que tenga instrucci&oacute;n PESO " & Format(kg, "0")) & _
                        "</table><br>" & TablaHtmlRango(pv, ini, fin, 13)
                    CrearCorreo CStr(pv.Range("B2").Value), CStr(pv.Range("B3").Value), _
                        "Solicitud Vehículo No " & numVeh & " Doc. Transporte " & docTransp & " " & Format(Date, "dd/mm/yyyy"), cuerpo, Nothing
                    hechas = hechas & vbCrLf & "Vehiculo " & numVeh & ": " & pv.Cells(r, 4).Value & " (" & rems.Count & " remisiones)"
                    Dim kk As Variant
                    For Each kk In rems.Keys: todas(kk) = "Vehiculo " & numVeh & " - " & pv.Cells(r, 4).Value: Next kk
                    pv.Cells(r, 6).Value = "PEDIDO"
                End If
            End If
            r = fin
        End If
    Next r

    If Len(omitidas) > 0 Then MsgBox "No se armaron estos vehiculos (corrija en " & HOJA_PREV_VEH & " y vuelva a ejecutar):" & omitidas, vbExclamation
    If todas.Count = 0 Then
        If Len(omitidas) = 0 Then MsgBox "No hay vehiculos con Pedir = SI.", vbInformation
        Exit Sub
    End If
    If MsgBox("Borradores creados:" & hechas & vbCrLf & vbCrLf & _
              "Despues de enviarlos, marcar estas " & todas.Count & " remisiones en CRUCE?", vbYesNo + vbQuestion) = vbYes Then
        MarcarEnCruceObs todas
    End If
End Sub

Private Function MaxDescargues(nombre As String) As Long
    Select Case UCase(Trim(nombre))
        Case "TURBO": MaxDescargues = 13
        Case "SENCILLO": MaxDescargues = 10
        Case "PATINETA", "TRACTOMULA", "MINIMULA": MaxDescargues = 1
        Case Else: MaxDescargues = 999
    End Select
End Function

' Punto de entrega = direccion (columna Direccion) normalizada; sin direccion, el cliente
Private Function PuntoEntrega(ws As Worksheet, fila As Variant) As String
    Dim s As String
    s = NormCiudad(CStr(ws.Cells(fila, C_DIR).Value))
    If Len(s) = 0 Then s = "CLIENTE " & NormCiudad(CStr(ws.Cells(fila, C_CLIENTE).Value))
    PuntoEntrega = NormCiudad(CStr(ws.Cells(fila, C_CIUDAD).Value)) & "|" & s
End Function

' Hay algun vehiculo donde quepan estos m3 y estos descargues?
Private Function Factible(m3 As Double, pts As Long) As Boolean
    Dim k As Long
    For k = 1 To UBound(vehNombre)
        If m3 <= vehCap(k) * TOPE_SATURACION And pts <= MaxDescargues(vehNombre(k)) Then Factible = True: Exit Function
    Next k
End Function

' Regla de rutas: si la zona tiene carga para CUCUTA, Cucuta y Bucaramanga van
' juntas en vehiculo(s) propio(s), sin otras ciudades. Si no hay carga para
' Cucuta, Bucaramanga se puede unir con Bogota y el resto de la zona.
Private Function CargasZona(ws As Worksheet, filas As Collection) As Collection
    Dim i As Long, ciudad As String, hayCucuta As Boolean
    For i = 1 To filas.Count
        If InStr(NormCiudad(CStr(ws.Cells(filas(i), C_CIUDAD).Value)), "CUCUTA") > 0 Then hayCucuta = True: Exit For
    Next i
    If Not hayCucuta Then Set CargasZona = ArmarCargas(ws, filas): Exit Function

    Dim ruta As New Collection, resto As New Collection, res As New Collection, c As Variant
    For i = 1 To filas.Count
        ciudad = NormCiudad(CStr(ws.Cells(filas(i), C_CIUDAD).Value))
        If InStr(ciudad, "CUCUTA") > 0 Or InStr(ciudad, "BUCARAMANGA") > 0 Then ruta.Add filas(i) Else resto.Add filas(i)
    Next i
    For Each c In ArmarCargas(ws, ruta): res.Add c: Next c
    If resto.Count > 0 Then
        For Each c In ArmarCargas(ws, resto): res.Add c: Next c
    End If
    Set CargasZona = res
End Function

' Divide las filas de una zona en la menor cantidad de cargas posible. Un punto de
' entrega nunca se parte. Cada carga debe caber en algun vehiculo por m3 y por
' maximo de descargues. Para cada cantidad de vehiculos se intenta primero
' mantener juntas las ciudades y, si no alcanza, repartir equilibrando la ocupacion.
Private Function ArmarCargas(ws As Worksheet, filas As Collection) As Collection
    Dim pts As Object, i As Long, key As String
    Set pts = CreateObject("Scripting.Dictionary")
    For i = 1 To filas.Count
        key = PuntoEntrega(ws, filas(i))
        If Not pts.Exists(key) Then pts.Add key, New Collection
        pts(key).Add filas(i)
    Next i

    ' Puntos ordenados de mayor a menor m3
    Dim np As Long, claves As Variant, pCiudad() As String, pM3() As Double, ord() As Long, a As Long, b As Long, t As Long
    np = pts.Count: claves = pts.Keys
    ReDim pCiudad(1 To np): ReDim pM3(1 To np): ReDim ord(1 To np)
    Dim f As Variant
    For a = 1 To np
        pCiudad(a) = Split(claves(a - 1), "|")(0)
        For Each f In pts(claves(a - 1)): pM3(a) = pM3(a) + Num(ws.Cells(f, C_CUB).Value): Next f
        ord(a) = a
    Next a
    For a = 1 To np - 1
        For b = a + 1 To np
            If pM3(ord(b)) > pM3(ord(a)) Then t = ord(a): ord(a) = ord(b): ord(b) = t
        Next b
    Next a

    Dim k As Long, modo As Long, asign() As Long, ok As Boolean
    ReDim asign(1 To np)
    For k = 1 To np
        For modo = 1 To 2
            ok = Repartir(k, modo, np, ord, pCiudad, pM3, asign)
            If ok Then Exit For
        Next modo
        If ok Then Exit For
    Next k
    If Not ok Then   ' algun punto no cabe en ningun vehiculo: uno por carga
        k = np
        For a = 1 To np: asign(a) = a: Next a
    End If

    Dim res As New Collection, carga As Collection, j As Long
    For j = 1 To k
        Set carga = New Collection
        For a = 1 To np
            If asign(a) = j Then
                For Each f In pts(claves(a - 1)): carga.Add f: Next f
            End If
        Next a
        If carga.Count > 0 Then res.Add carga
    Next j
    Set ArmarCargas = res
End Function

' Intenta repartir los puntos en k cargas. modo 1 = prefiere la carga que ya
' tiene la misma ciudad; modo 2 = solo equilibra la ocupacion.
Private Function Repartir(k As Long, modo As Long, np As Long, ord() As Long, pCiudad() As String, _
                          pM3() As Double, asign() As Long) As Boolean
    Dim lM3() As Double, lPts() As Long, lCiud() As String, a As Long, p As Long, j As Long
    Dim mejor As Long, mejorUso As Double, u As Double, mismaCiudad As Boolean, mejorMisma As Boolean
    ReDim lM3(1 To k): ReDim lPts(1 To k): ReDim lCiud(1 To k)
    For a = 1 To np
        p = ord(a)
        mejor = 0
        For j = 1 To k
            u = Uso(lM3(j) + pM3(p), lPts(j) + 1)
            If u < 1E+30 Then
                mismaCiudad = (modo = 1 And InStr(lCiud(j), "|" & pCiudad(p) & "|") > 0)
                If mejor = 0 Then
                    mejor = j: mejorUso = u: mejorMisma = mismaCiudad
                ElseIf (mismaCiudad And Not mejorMisma) Or (mismaCiudad = mejorMisma And u < mejorUso) Then
                    mejor = j: mejorUso = u: mejorMisma = mismaCiudad
                End If
            End If
        Next j
        If mejor = 0 Then Exit Function
        lM3(mejor) = lM3(mejor) + pM3(p)
        lPts(mejor) = lPts(mejor) + 1
        If InStr(lCiud(mejor), "|" & pCiudad(p) & "|") = 0 Then lCiud(mejor) = lCiud(mejor) & "|" & pCiudad(p) & "|"
        asign(p) = mejor
    Next a
    Repartir = True
End Function

' Ocupacion del vehiculo que mejor recibe la carga (mayor entre % m3 y % descargues);
' 1E+31 si no cabe en ninguno.
Private Function Uso(m3 As Double, pts As Long) As Double
    Dim k As Long, u As Double
    Uso = 1E+31
    For k = 1 To UBound(vehNombre)
        If m3 <= vehCap(k) * TOPE_SATURACION And pts <= MaxDescargues(vehNombre(k)) Then
            u = Application.Max(m3 / vehCap(k), pts / MaxDescargues(vehNombre(k)))
            If u < Uso Then Uso = u
        End If
    Next k
End Function

'==============================================================================
' Marcar manualmente en CRUCE las filas visibles de CUBICAR HOY
'==============================================================================
Public Sub MarcarRemisionesCubicadas()
    Dim ws As Worksheet, i As Long, ultima As Long, rems As Object, r As String
    Set ws = HojaDetalle(): If ws Is Nothing Then Exit Sub
    Set rems = CreateObject("Scripting.Dictionary")
    ultima = ws.Cells(ws.Rows.Count, C_REM).End(xlUp).Row
    For i = 2 To ultima
        r = Clave(ws.Cells(i, C_REM).Value)
        If Not ws.Rows(i).Hidden And Len(r) > 0 Then rems(r) = True
    Next i
    If rems.Count = 0 Then Exit Sub
    If MsgBox("Marcar en CRUCE las " & rems.Count & " remisiones visibles?", vbYesNo + vbQuestion) = vbYes Then
        MarcarEnCruce rems, "Marcada manual"
    End If
End Sub

Private Sub MarcarEnCruce(rems As Object, obs As String)
    Dim d As Object, k As Variant
    Set d = CreateObject("Scripting.Dictionary")
    For Each k In rems.Keys: d(k) = obs: Next k
    MarcarEnCruceObs d
End Sub

' rems: remision -> observacion
Private Sub MarcarEnCruceObs(rems As Object)
    Dim wsC As Worksheet, dCruce As Object, fila As Long, k As Variant, n As Long
    Set wsC = ThisWorkbook.Worksheets("CRUCE")
    Set dCruce = Diccionario(wsC, 1, 1, 2)
    fila = wsC.Cells(wsC.Rows.Count, 1).End(xlUp).Row + 1
    For Each k In rems.Keys
        If Not dCruce.Exists(CStr(k)) Then
            wsC.Cells(fila, 1).Value = CDbl(k)
            wsC.Cells(fila, 2).Value = rems(k) & " - " & Format(Now, "dd/mm/yyyy hh:nn")
            fila = fila + 1: n = n + 1
        End If
    Next k
    MsgBox n & " remisiones agregadas a CRUCE.", vbInformation
End Sub

' enviadas: remision -> texto del corte
Private Sub RegistrarEnviadasBodega(enviadas As Object)
    Dim h As Worksheet, fila As Long, k As Variant, ws As Worksheet, i As Long, ultima As Long
    Set h = HojaHistorial()
    fila = h.Cells(h.Rows.Count, 1).End(xlUp).Row + 1
    For Each k In enviadas.Keys
        h.Cells(fila, 1).Value = CDbl(k)
        h.Cells(fila, 2).Value = enviadas(k)
        h.Cells(fila, 3).Value = Now
        fila = fila + 1
    Next k
    ' Marcar en CUBICAR HOY para que no vuelvan a salir en este mismo dia
    Set ws = HojaDetalle()
    If Not ws Is Nothing Then
        ultima = ws.Cells(ws.Rows.Count, C_REM).End(xlUp).Row
        For i = 2 To ultima
            If enviadas.Exists(Clave(ws.Cells(i, C_REM).Value)) Then ws.Cells(i, C_NUEVO).Value = "NO"
        Next i
    End If
    MsgBox enviadas.Count & " remisiones registradas en " & HOJA_HIST & ".", vbInformation
End Sub

Private Sub CrearCorreo(para As String, cc As String, asunto As String, cuerpoHtml As String, adjuntos As Collection)
    Dim ol As Object, mail As Object, i As Long
    On Error Resume Next
    Set ol = CreateObject("Outlook.Application")
    On Error GoTo 0
    If ol Is Nothing Then MsgBox "No se pudo abrir Outlook.", vbCritical: Exit Sub

    Set mail = ol.CreateItem(0)
    mail.Display                       ' primero se muestra para conservar la firma de Outlook
    mail.To = para
    mail.CC = cc
    mail.Subject = asunto
    mail.HTMLBody = "<div style='font-family:Calibri;font-size:11pt'>" & cuerpoHtml & "</div>" & mail.HTMLBody
    If Not adjuntos Is Nothing Then
        For i = 1 To adjuntos.Count
            mail.Attachments.Add adjuntos(i)
        Next i
    End If
End Sub

' Tabla HTML del correo a partir de la vista previa: encabezado en la fila ini,
' lineas de ini+1 a fin, columnas 1..nCols; usa el texto tal como se ve en Excel.
Private Function TablaHtmlRango(sh As Worksheet, ini As Long, fin As Long, nCols As Long) As String
    Dim h As String, r As Long, c As Long, alin As String
    h = "<table border='1' cellspacing='0' cellpadding='3' style='border-collapse:collapse;font-family:Calibri;font-size:10pt'><tr>"
    For c = 1 To nCols
        h = h & "<th style='background:#FFFF00'>" & Html(CStr(sh.Cells(ini, c).Value)) & "</th>"
    Next c
    h = h & "</tr>"
    For r = ini + 1 To fin
        h = h & "<tr>"
        For c = 1 To nCols
            alin = IIf(c = 8 Or c = 9 Or c = 13, " align='right'", "")
            h = h & "<td" & alin & ">" & Html(sh.Cells(r, c).Text) & "</td>"
        Next c
        h = h & "</tr>"
    Next r
    TablaHtmlRango = h & "</table>"
End Function

' Agrupa las filas por Codigo SAP (o por direccion si no hay codigo), en el orden
' en que aparece cada cliente; dentro de cada grupo, por numero de remision.
Private Function OrdenAgrupado(ws As Worksheet, filas As Collection) As Collection
    Dim grupos As Object, i As Long, g As String, k As Variant, res As New Collection
    Dim a As Long, b As Long, n As Long, t As Variant, arr() As Variant
    Set grupos = CreateObject("Scripting.Dictionary")
    For i = 1 To filas.Count
        g = Clave(ws.Cells(filas(i), C_CODSAP).Value)
        If Len(g) = 0 Then g = "DIR|" & NormCiudad(CStr(ws.Cells(filas(i), C_DIR).Value))
        If Not grupos.Exists(g) Then grupos.Add g, New Collection
        grupos(g).Add filas(i)
    Next i
    For Each k In grupos.Keys
        n = grupos(k).Count
        ReDim arr(1 To n)
        For a = 1 To n: arr(a) = grupos(k)(a): Next a
        For a = 2 To n
            t = arr(a): b = a - 1
            Do While b >= 1
                If Num(ws.Cells(arr(b), C_REM).Value) <= Num(ws.Cells(t, C_REM).Value) Then Exit Do
                arr(b + 1) = arr(b): b = b - 1
            Loop
            arr(b + 1) = t
        Next a
        For a = 1 To n: res.Add arr(a): Next a
    Next k
    Set OrdenAgrupado = res
End Function

' "1 Corte Octubre 14" + "PIRELLI TYRE" -> "PRIMER PIRELLI OCTUBRE 14"
Private Function AsuntoCorte(corte As String, empresa As String) As String
    Dim p As Variant, marca As String
    marca = UCase(Split(Trim(empresa) & " ", " ")(0))
    p = Split(Application.WorksheetFunction.Trim(corte), " ")
    If UBound(p) >= 3 Then
        AsuntoCorte = Ordinal(CStr(p(0))) & " " & marca & " " & UCase(p(2)) & " " & p(3)
    Else
        AsuntoCorte = UCase(corte) & " " & marca
    End If
End Function

' "1 Corte Octubre 14" -> "primer corte del 14 de octubre"
Private Function TextoCorte(corte As String) As String
    Dim p As Variant
    p = Split(Application.WorksheetFunction.Trim(corte), " ")
    If UBound(p) >= 3 Then
        TextoCorte = LCase(Ordinal(CStr(p(0)))) & " corte del " & p(3) & " de " & LCase(p(2))
    Else
        TextoCorte = "corte " & corte
    End If
End Function

Private Function Ordinal(n As String) As String
    Select Case Trim(n)
        Case "1": Ordinal = "PRIMER"
        Case "2": Ordinal = "SEGUNDO"
        Case "3": Ordinal = "TERCER"
        Case "4": Ordinal = "CUARTO"
        Case "5": Ordinal = "QUINTO"
        Case Else: Ordinal = n
    End Select
End Function

' Lee los TXT de la carpeta y los indexa por el numero que aparece tras "Remision:"
' (si hay varios para la misma remision se toma el mas reciente)
Private Function ArchivosTxtPorRemision(carpeta As String) As Object
    Dim d As Object, fechas As Object, re As Object, f As String, ruta As String, s As String, ff As Integer, m As Object
    Set d = CreateObject("Scripting.Dictionary")
    Set fechas = CreateObject("Scripting.Dictionary")
    Set re = CreateObject("VBScript.RegExp")
    re.Pattern = "Remisi.n:\s*(\d{8,12})"
    re.IgnoreCase = True
    If Right(carpeta, 1) <> "\" Then carpeta = carpeta & "\"
    f = Dir(carpeta & "*.txt")
    Do While Len(f) > 0
        ruta = carpeta & f
        ff = FreeFile
        Open ruta For Binary Access Read As #ff
        s = Space$(Application.Min(LOF(ff), 4000))
        Get #ff, , s
        Close #ff
        If re.Test(s) Then
            Set m = re.Execute(s)(0)
            If Not d.Exists(m.SubMatches(0)) Then
                d.Add m.SubMatches(0), ruta: fechas.Add m.SubMatches(0), FileDateTime(ruta)
            ElseIf FileDateTime(ruta) > fechas(m.SubMatches(0)) Then
                d(m.SubMatches(0)) = ruta: fechas(m.SubMatches(0)) = FileDateTime(ruta)
            End If
        End If
        f = Dir()
    Loop
    Set ArchivosTxtPorRemision = d
End Function

End Function

Private Function IndiceSugerido(m3 As Double, Optional pts As Long = 1) As Long
    Dim k As Long, mejor As Long
    For k = 1 To UBound(vehNombre)
        If m3 / vehCap(k) <= TOPE_SATURACION And pts <= MaxDescargues(vehNombre(k)) Then
            If mejor = 0 Then
                mejor = k
            ElseIf vehCap(k) < vehCap(mejor) Then
                mejor = k
            End If
        End If
    Next k
    If mejor = 0 Then
        For k = 1 To UBound(vehNombre)
            If pts <= MaxDescargues(vehNombre(k)) Then
                If mejor = 0 Then
                    mejor = k
                ElseIf vehCap(k) > vehCap(mejor) Then
                    mejor = k
                End If
            End If
        Next k
    End If
    If mejor = 0 Then mejor = 1
    IndiceSugerido = mejor
End Function

Private Sub LeerVehiculos()
    Dim ws As Worksheet, i As Long, n As Long, etq As String
    Set ws = ThisWorkbook.Worksheets("CUBICAJE BODEGA")
    ReDim vehNombre(1 To 4): ReDim vehCap(1 To 4): ReDim vehEtiqueta(1 To 4)
    For i = 3 To 6
        If Len(Trim(CStr(ws.Cells(i, 2).Value))) > 0 And Num(ws.Cells(i, 3).Value) > 0 Then
            n = n + 1
            vehNombre(n) = Trim(CStr(ws.Cells(i, 2).Value))
            vehCap(n) = Num(ws.Cells(i, 3).Value)
            etq = Trim(Replace(LCase(CStr(ws.Cells(i, 1).Value)), "m3", ""))   ' "42 m3" -> "42"
            vehEtiqueta(n) = vehNombre(n) & IIf(Len(etq) > 0, " " & etq, "")
        End If
    Next i
    If n = 0 Then Err.Raise vbObjectError + 1, , "No se encontraron vehiculos en CUBICAJE BODEGA!B3:C6"
    ReDim Preserve vehNombre(1 To n): ReDim Preserve vehCap(1 To n): ReDim Preserve vehEtiqueta(1 To n)
End Sub

Private Function Diccionario(ws As Worksheet, colClave As Long, colValor As Long, filaIni As Long) As Object
    Dim d As Object, v As Variant, i As Long, ultima As Long, k As String
    Set d = CreateObject("Scripting.Dictionary")
    ultima = ws.Cells(ws.Rows.Count, colClave).End(xlUp).Row
    If ultima >= filaIni Then
        v = ws.Range(ws.Cells(filaIni, 1), ws.Cells(ultima, Application.Max(colClave, colValor))).Value
        For i = 1 To UBound(v, 1)
            k = Clave(v(i, colClave))
            If Len(k) > 0 And Not d.Exists(k) Then d.Add k, v(i, colValor)
        Next i
    End If
    Set Diccionario = d
End Function

Private Function DiccionarioCiudades(ws As Worksheet, colCiudad As Long, colZona As Long, filaIni As Long) As Object
    Dim d As Object, v As Variant, i As Long, ultima As Long, k As String
    Set d = CreateObject("Scripting.Dictionary")
    ultima = ws.Cells(ws.Rows.Count, colCiudad).End(xlUp).Row
    v = ws.Range(ws.Cells(filaIni, colCiudad), ws.Cells(ultima, colZona)).Value
    For i = 1 To UBound(v, 1)
        k = NormCiudad(CStr(v(i, 1)))
        If Len(k) > 0 And Not d.Exists(k) Then d.Add k, Trim(CStr(v(i, colZona - colCiudad + 1)))
    Next i
    Set DiccionarioCiudades = d
End Function

' Busca la columna por nombre exacto en la fila 1 (primera coincidencia)
Private Function Col(datos As Variant, nombre As String) As Long
    Dim j As Long
    For j = 1 To UBound(datos, 2)
        If Trim(CStr(datos(1, j))) = nombre Then Col = j: Exit Function
    Next j
    Err.Raise vbObjectError + 2, , "El export no tiene la columna '" & nombre & "'"
End Function

' Normaliza codigos: 2915059484, "2915059484" y 2915059484.0 dan la misma clave
Private Function Clave(v As Variant) As String
    If IsError(v) Or IsEmpty(v) Then Exit Function
    Dim s As String: s = Trim(CStr(v))
    If Len(s) = 0 Then Exit Function
    If IsNumeric(s) Then Clave = Format(CDbl(s), "0") Else Clave = UCase(s)
End Function

Private Function Num(v As Variant) As Double
    If IsError(v) Or IsEmpty(v) Then Exit Function
    If IsNumeric(v) Then Num = CDbl(v)
End Function

' "BOGOTA D.C." y "BOGOTA, D.C." quedan iguales; tambien quita tildes
Private Function NormCiudad(s As String) As String
    Dim r As String, a As Variant, b As Variant, i As Long
    r = UCase(Trim(s))
    a = Array("Á", "É", "Í", "Ó", "Ú", "Ü", ",", ".")
    b = Array("A", "E", "I", "O", "U", "U", " ", " ")
    For i = 0 To UBound(a): r = Replace(r, a(i), b(i)): Next i
    Do While InStr(r, "  ") > 0: r = Replace(r, "  ", " "): Loop
    NormCiudad = Trim(r)
End Function

Private Function Ordenadas(d As Object) As Variant
    Dim a As Variant, i As Long, j As Long, t As Variant
    a = d.Keys
    For i = LBound(a) To UBound(a) - 1
        For j = i + 1 To UBound(a)
            If a(j) < a(i) Then t = a(i): a(i) = a(j): a(j) = t
        Next j
    Next i
    Ordenadas = a
End Function
